import json, urllib.request
base='http://127.0.0.1:3000'
def request(path, method='GET', payload=None, timeout=180):
    data=None if payload is None else json.dumps(payload).encode()
    req=urllib.request.Request(base+path,data=data,method=method,headers={'Content-Type':'application/json'} if data else {})
    with urllib.request.urlopen(req,timeout=timeout) as r:
        body=r.read()
        try: parsed=json.loads(body)
        except Exception: parsed=body
        print(method,path,r.status,r.headers.get_content_type(),len(body))
        return parsed,r.headers
for job in ['raster','java','dataset','train_and_predict']:
    d,_=request('/api/pipeline/preflight/'+job)
    print(' preflight:', d)
result,_=request('/api/model/train','POST',{'split_ratio':0.2,'folds':3,'seed':42},timeout=180)
res=result.get('result',{})
print(' training provenance:',res.get('provenance'),'demo:',res.get('demo'),'validation:',res.get('validation'))
for m in res.get('metrics',[]):
    print(' model:',m.get('algo'),'available:',m.get('available',True),'f1:',m.get('f1'),'rmse:',m.get('rmse'),'kfold:',(m.get('kfold') or {}).get('available'))
metrics,_=request('/api/metrics')
print(' metrics rows:',len(metrics.get('metrics',[])),'demo:',metrics.get('demo'),'status:',metrics.get('status'))
pred,_=request('/api/model/predict','POST',{})
print(' predictions:',pred.get('persisted'),'demo:',pred.get('demo'))
explain,_=request('/api/explain/DEMO-01')
print(' explanation fields:',sorted(explain.keys()),'contributions:',len(explain.get('contributions',[])))
for path in ['/api/export/csv','/api/export/pdf']:
    _,headers=request(path)
    print(' export disposition:',headers.get('Content-Disposition'))
thresholds,_=request('/api/settings/thresholds','PUT',{'low_max':0.25,'moderate_max':0.5,'high_max':0.75,'temperature_max':35})
print(' thresholds persisted:',thresholds)
health,_=request('/api/health')
print(' health:',health.get('ok'),health.get('database',{}).get('engine'),health.get('rasters',{}).get('present_count'))
with urllib.request.urlopen(base+'/openapi.json') as r:
    api=json.load(r)
print(' api paths:',', '.join(sorted(api.get('paths',{}))))
