import ee
import geemap
import os

# 1. Authenticate and Initialize Earth Engine
project_id = 'dataset-510306'

try:
    ee.Initialize(project=project_id)
except Exception as e:
    ee.Authenticate()
    ee.Initialize(project=project_id)

os.makedirs("data/raw", exist_ok=True)

# 2. Define Expanded Nagpur Bounding Box (~20km radius from Zero Mile)
nagpur_roi = ee.Geometry.Rectangle([78.89, 20.96, 79.29, 21.33])

print("Fetching Sentinel-2 Greenery (NDVI)...")
s2 = ee.ImageCollection('COPERNICUS/S2_SR_HARMONIZED') \
    .filterBounds(nagpur_roi) \
    .filterDate('2023-04-01', '2023-05-31') \
    .filterMetadata('CLOUDY_PIXEL_PERCENTAGE', 'less_than', 10) \
    .median()
ndvi = s2.normalizedDifference(['B8', 'B4']).rename('NDVI')
geemap.ee_export_image(ndvi, filename="data/raw/nagpur_ndvi.tif", scale=10, region=nagpur_roi)

print("Fetching Landsat 8 Heat Map (LST)...")
l8 = ee.ImageCollection("LANDSAT/LC08/C02/T1_L2") \
    .filterBounds(nagpur_roi) \
    .filterDate('2023-04-01', '2023-05-31') \
    .median()
# Convert thermal band to Celsius
lst = l8.select('ST_B10').multiply(0.00341802).add(149.0).subtract(273.15).rename('LST')
geemap.ee_export_image(lst, filename="data/raw/nagpur_lst.tif", scale=30, region=nagpur_roi)

print("Fetching SRTM Elevation (Slope)...")
srtm = ee.Image('USGS/SRTMGL1_003').clip(nagpur_roi)
slope = ee.Terrain.slope(srtm).rename('Slope')
geemap.ee_export_image(slope, filename="data/raw/nagpur_slope.tif", scale=30, region=nagpur_roi)

print("Fetching WorldPop Population Density...")
pop = ee.ImageCollection("WorldPop/GP/100m/pop") \
    .filterBounds(nagpur_roi) \
    .filterDate('2020-01-01', '2021-01-01') \
    .mean().rename('Population')
geemap.ee_export_image(pop, filename="data/raw/nagpur_pop.tif", scale=100, region=nagpur_roi)

print("All satellite rasters downloaded successfully to data/raw/")
FILE 2: 2_build_real_dataset.py
python
import os
import osmnx as ox
import geopandas as gpd
import pandas as pd
from rasterstats import zonal_stats
import warnings
from sklearn.preprocessing import MinMaxScaler

warnings.filterwarnings('ignore')

print("1. Downloading Nagpur Road Network (Vector data)...")
# CHANGED: dist=5000 is now dist=20000 to include outer highways and ring roads
G = ox.graph_from_point((21.1458, 79.0882), dist=20000, network_type='all')
nodes, edges = ox.graph_to_gdfs(G)
edges = edges.reset_index()

# Convert to UTM Zone 44N (meters) to draw the 10m buffer
edges_proj = edges.to_crs(epsg=32644)
buffered_geoms = edges_proj.geometry.buffer(10)

# THE FIX: Convert the buffered polygons BACK to EPSG:4326 (degrees) 
# so they match the satellite images downloaded from Earth Engine.
edges_buffered_gdf = gpd.GeoDataFrame(geometry=buffered_geoms, crs=32644)
edges_ready_for_raster = edges_buffered_gdf.to_crs(epsg=4326)

print("2. Extracting Real Satellite Data for every street (Raster to Vector)...")
def extract_raster_stats(raster_path):
    if not os.path.exists(raster_path):
        print(f"  [!] Missing {raster_path}. Did the download script finish?")
        return [0] * len(edges)
    
    # We now use 'edges_ready_for_raster' which is in degrees
    stats = zonal_stats(edges_ready_for_raster, raster_path, stats="mean", all_touched=True)
    return [s['mean'] if s['mean'] is not None else 0 for s in stats]

edges['lst'] = extract_raster_stats("data/raw/nagpur_lst.tif")
edges['ndvi'] = extract_raster_stats("data/raw/nagpur_ndvi.tif")
edges['slope'] = extract_raster_stats("data/raw/nagpur_slope.tif")
edges['population'] = extract_raster_stats("data/raw/nagpur_pop.tif")

edges['pm'] = 65.0 

print("3. Calculating AHP Suitability Scores...")
df = pd.DataFrame(edges[['osmid', 'lst', 'ndvi', 'pm', 'population', 'slope']])
df = df.rename(columns={'osmid': 'edge_id'})
df['edge_id'] = df['edge_id'].astype(str)

features = ["lst", "ndvi", "pm", "population", "slope"]
scaler = MinMaxScaler()
df_norm = pd.DataFrame(scaler.fit_transform(df[features]), columns=features)

df["composite_suitability_score"] = (
    (1 - df_norm["lst"]) * 0.35 +
    df_norm["ndvi"] * 0.25 +
    (1 - df_norm["pm"]) * 0.20 +
    (1 - df_norm["population"]) * 0.10 +
    (1 - df_norm["slope"]) * 0.10
)

os.makedirs("data/processed", exist_ok=True)
df.to_csv("data/processed/edge_suitability.csv", index=False)
print("Success! Real dataset saved to data/processed/edge_suitability.csv")
FILE 3: GisImageDownloader.java
java
import java.io.InputStream;
import java.net.URL;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.Paths;
import java.nio.file.StandardCopyOption;

public class GisImageDownloader {
    
    // The target directory your ML scripts are expecting
    private static final String DATA_DIR = "data/raw/";
    
    // The exact GIS GeoTIFF rasters required for the GeoCool project
    private static final String[] RASTER_FILES = {
        "nagpur_lst.tif",
        "nagpur_ndvi.tif",
        "nagpur_slope.tif",
        "nagpur_pop.tif"
    };

    public static void main(String[] args) {
        System.out.println("Initializing GIS Raster Download Pipeline...");
        
        // In a live production environment, this base URL points to an authenticated 
        // Google Earth Engine REST endpoint or a cloud bucket hosting the exported images.
        String cloudStorageUrl = "https://geocool-satellite-data.s3.amazonaws.com/nagpur/5km_radius/";

        try {
            Path dirPath = Paths.get(DATA_DIR);
            if (!Files.exists(dirPath)) {
                Files.createDirectories(dirPath);
                System.out.println("Created missing directory: " + DATA_DIR);
            }

            for (String fileName : RASTER_FILES) {
                System.out.println("Fetching -> " + fileName);
                URL url = new URL(cloudStorageUrl + fileName);
                
                try (InputStream in = url.openStream()) {
                    Path destination = Paths.get(DATA_DIR + fileName);
                    // This overwrites any partial or corrupted files from a previous run
                    Files.copy(in, destination, StandardCopyOption.REPLACE_EXISTING);
                    System.out.println("Successfully saved to: " + destination.toAbsolutePath());
                } catch (Exception e) {
                    System.err.println("Error downloading " + fileName + ": " + e.getMessage());
                }
            }
            
            System.out.println("All GIS image assets provisioned.");
            
        } catch (Exception e) {
            System.err.println("Critical pipeline failure: " + e.getMessage());
        }
    }
}
